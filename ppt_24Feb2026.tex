% Slide 2 - Problem Statement
\begin{frame}{Problem Statement}
\begin{itemize}
    \item Difficulty in managing daily tasks efficiently
    \item Manual tracking leads to missed deadlines
    \item Existing platforms are feature-heavy or complex
    \item Need for a simple and lightweight solution
\end{itemize}
\end{frame}

% Slide 3 - Proposed Solution 
\begin{frame}{Proposed Solution}
\begin{itemize}
    \item Web-based Task Management System
    \item Secure user authentication
    \item Create, update, delete tasks (CRUD)
    \item Priority-based categorization
    \item Simple and user-friendly interface
\end{itemize}
\end{frame}

% Slide 4 - Tech Stack 
\begin{frame}{Technology Stack}
\begin{itemize}
    \item \textbf{Frontend:} HTML, CSS, JavaScript
    \item \textbf{Backend:} FastAPI (Python-based framework)
    \item \textbf{Database:} SQLite3
    \item \textbf{Security:} Password Hashing (Passlib/Bcrypt)
    \item \textbf{Configuration:} Environment Variables (.env)
\end{itemize}
\end{frame}

% Slide 5 - System Architecture
\begin{frame}{System Architecture}

\begin{block}{Frontend}
HTML, CSS, JavaScript – User Interface
\end{block}

\begin{block}{Backend}
FastAPI – API handling, authentication, logic
\end{block}

\begin{block}{Database}
SQLite3 – Task \& User data storage
\end{block}
\end{frame}

% Slide 6 - Security
\begin{frame}{Security Implementation}
\begin{itemize}
    \item Secure user authentication system
    \item Password hashing before database storage
    \item Environment variables for secret configuration
    \item Database file excluded from version control
    \item Input validation at backend layer
\end{itemize}
\end{frame}
