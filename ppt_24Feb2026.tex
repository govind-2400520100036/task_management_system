% Slide 2 - Problem Statement
\begin{frame}{Problem Statement}
\begin{itemize}
    \item Difficulty in managing daily tasks efficiently
    \item Manual tracking leads to missed deadlines
    \item Existing platforms are feature-heavy or complex
    \item Need for a simple and lightweight solution
\end{itemize}
\end{frame}

% Slide 3 - Proposed Solution 
\begin{frame}{Proposed Solution}
\begin{itemize}
    \item Web-based Task Management System
    \item Secure user authentication
    \item Create, update, delete tasks (CRUD)
    \item Priority-based categorization
    \item Simple and user-friendly interface
\end{itemize}
\end{frame}

% Slide 4 - Tech Stack 
\begin{frame}{Technology Stack}
\begin{itemize}
    \item Frontend: HTML, CSS, JavaScript
    \item Backend: FastAPI (Python-based framework)
    \item Database: SQLite3
    \item Security: Password Hashing
    \item Configuration: Environment Variables
\end{itemize}
\end{frame}

% Slide 7 - Current Status
\begin{frame}{Current Status}
\begin{itemize}
   \item User Registration and Login module completed
    \item Task creation and deletion implemented
    \item Priority-based task categorization
    \item Search functionality integrated
    \item Backend API using FastAPI connected to SQLite
    \item Secure password hashing applied
\end{itemize}
\end{frame}

% Slide 8 - Future Scope
\begin{frame}{Future Scope}
\begin{itemize}
    \item Cloud deployment for remote access
    \item Role-based user authentication
    \item Real-time task updates
    \item Mobile responsive interface
    \item Integration with AI-based task prioritization
\end{itemize}
\end{frame}
